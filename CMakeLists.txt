cmake_minimum_required(VERSION 3.18)
project(cuda_vision_pipeline LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 86)
find_package(OpenCV REQUIRED)
find_package(Threads REQUIRED)
set(ORT_ROOT "/home/ragul/libs/onnxruntime")
file(GLOB NV_LIB_DIRS "/home/ragul/.local/lib/python3.10/site-packages/nvidia/*/lib")
include_directories(${OpenCV_INCLUDE_DIRS} ${ORT_ROOT}/include include)
link_directories(${ORT_ROOT}/lib ${CMAKE_BINARY_DIR})
add_executable(vision_pipeline src/main.cpp src/vision_kernels.cu)
target_link_libraries(vision_pipeline ${OpenCV_LIBS} Threads::Threads onnxruntime)
set_target_properties(vision_pipeline PROPERTIES BUILD_RPATH "${ORT_ROOT}/lib;${NV_LIB_DIRS}")
